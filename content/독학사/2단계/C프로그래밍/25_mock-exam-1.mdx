### 10~12번 — 함수와 변수의 범위 (09편)

```c
#include <stdio.h>

void addTen(int x) {
    x = x + 10;
}

int main(void) {
    int n = 5;
    addTen(n);
    printf("%d\n", n);
    return 0;
}
```

```c
#include <stdio.h>

int x = 100;

void changeX(void) {
    int x = 5;
    x = x + 1;
}

int main(void) {
    changeX();
    printf("%d\n", x);
    return 0;
}
```

### 13~15번 — 배열·문자열·포인터 기초 (10, 11편)

```c
#include <stdio.h>

int main(void) {
    int arr[5] = {10, 20, 30, 40, 50};
    int sum = 0;
    for (int i = 1; i < 4; i++) {
        sum += arr[i];
    }
    printf("%d
", sum);
    return 0;
}
```

### 16~18번 — 구조체·전처리기 (13, 14편)

```c
#include <stdio.h>

struct Book {
    char title[20];
    int price;
};

int main(void) {
    struct Book b1 = {"C Programming", 15000};
    struct Book b2 = b1;
    b2.price = 20000;
    printf("%d %d\n", b1.price, b2.price);
    return 0;
}
```
