### 7~9번 — 조건문·반복문 (09편)

```c
#include <stdio.h>

int main(void) {
    int score = 75;
    char grade;

    if (score >= 90) grade = 'A';
    else if (score >= 80) grade = 'B';
    else if (score >= 70) grade = 'C';
    else grade = 'F';

    printf("%c\n", grade);
    return 0;
}
```

```c
#include <stdio.h>

int main(void) {
    int x = 2;
    int sum = 0;

    switch (x) {
        case 1:
            sum += 1;
        case 2:
            sum += 2;
        case 3:
            sum += 3;
            break;
        default:
            sum += 100;
    }

    printf("%d\n", sum);
    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *p = (int *)malloc(3 * sizeof(int));
    p[0] = 1;
    p[1] = 2;
    p[2] = 3;
    printf("%d\n", p[0] + p[1] + p[2]);
    free(p);
    return 0;
}
```

    return 0;
}
