### 1~3번 — 포인터 심화: 함수 포인터·포인터 배열 (12, 13편)

```c
#include <stdio.h>

int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }

int main(void) {
    int (*fp)(int, int) = add;
    int result = fp(10, 3);
    fp = sub;
    result += fp(10, 3);
    printf("%d\n", result);
    return 0;
}
```

### 7~9번 — 구조체 심화: 구조체 포인터·중첩 구조체·동적 배열 (14, 18편)

```c
struct Point {
    int x, y;

int main(void) {
    struct Point p1 = {3, 4};

    ptr->x = ptr->x + 10;
    (*ptr).y = (*ptr).y + 20;
    printf("%d %d\n", p1.x, p1.y);
    return 0;
}
```

```c
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

int main(void) {
    struct Node *head = (struct Node *)malloc(sizeof(struct Node));
    head->data = 1;
    head->next = (struct Node *)malloc(sizeof(struct Node));
    head->next->data = 2;
    head->next->next = NULL;

    printf("%d %d\n", head->data, head->next->data);

    free(head->next);
    free(head);
    return 0;
}
```
