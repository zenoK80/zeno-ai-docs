### 1~3번 — 포인터 심화: 함수 포인터·포인터 배열 (11, 12편)

```c
#include <stdio.h>

int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }

int main(void) {
    int (*fp)(int, int) = add;
    int result = fp(10, 3);
    fp = sub;
    result += fp(10, 3);
    printf("%d\n", result);
    return 0;
}
```

### 21~22번 — 통합 문제 (17, 19편)

```c

struct Node {
    int data;
    struct Node *next;
};

int main(void) {
    struct Node *head = (struct Node *)malloc(sizeof(struct Node));
    head->data = 1;
    head->next = (struct Node *)malloc(sizeof(struct Node));
    head->next->data = 2;
    head->next->next = NULL;

    printf("%d %d\n", head->data, head->next->data);

    free(head->next);
    free(head);
    return 0;
}
```
