```java
    Parent() {
        show();
    }

    void show() {
        System.out.println("Parent show");
    }

    int value = 10;

    @Override
    void show() {
        System.out.println("Child show, value = " + value);
    }

    public static void main(String[] args) {
        new Child();
    }
}
```
