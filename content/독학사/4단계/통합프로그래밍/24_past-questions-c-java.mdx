### 문제 9

```c
#include <stdio.h>

struct Point {
    int x;
    int y;
};

void movePoint(struct Point *p, int dx, int dy) {
    p->x += dx;
    p->y += dy;
}

int main(void) {
    struct Point p1 = {1, 1};
    movePoint(&p1, 5, 5);
    printf("%d %d\n", p1.x, p1.y);
    return 0;
}
```
